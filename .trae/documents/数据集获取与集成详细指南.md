## 二、RESCO基准数据集详细方案

### 2.1 RESCO数据集概述

**RESCO (Reinforcement Learning Benchmarks for Traffic Signal Control)** 是专门为交通信号控制强化学习研究设计的基准数据集。

**优势**:
- ✅ SUMO-RL原生支持，开箱即用
- ✅ 包含多种路网拓扑结构
- ✅ 已有论文基准结果可对比
- ✅ 数据格式标准，无需转换
- ✅ 涵盖单Agent和多Agent场景

由于LuST规模较大，建议进行子集提取：

```python
import sumolib
import xml.etree.ElementTree as ET

def extract_lust_subset(
    net_file, 
    rou_file, 
    output_dir,
    time_range=(0, 3600),  # 提取1小时数据
    region=None             # 可选：提取特定区域
):
    """
    从LuST场景提取子集
    
    Args:
        net_file: 路网文件路径
        rou_file: 路由文件路径
        output_dir: 输出目录
        time_range: 时间范围 (begin, end)
        region: 区域范围 (x_min, y_min, x_max, y_max)
    """
    
    # 1. 读取路网
    net = sumolib.net.readNet(net_file)
    
    # 2. 提取区域子图
    if region:
        edges_in_region = []
        for edge in net.getEdges():
            # 检查边是否在区域内
            shape = edge.getShape()
            if is_in_region(shape, region):
                edges_in_region.append(edge.getID())
    
    # 3. 过滤路由
    tree = ET.parse(rou_file)
    root = tree.getroot()
    
    for vehicle in root.findall('vehicle'):
        depart = float(vehicle.get('depart'))
        # 移除时间范围外的车辆
        if depart < time_range[0] or depart > time_range[1]:
            root.remove(vehicle)
    
    # 4. 保存子集
    tree.write(f'{output_dir}/subset.rou.xml')
```

```python
import pandas as pd
import xml.etree.ElementTree as ET

def ngsim_to_sumo(
    ngsim_csv,
    output_net,
    output_rou,
    sample_rate=10  # 采样率（秒）
):
    """
    将NGSIM数据转换为SUMO格式
    
    Args:
        ngsim_csv: NGSIM CSV文件
        output_net: 输出路网文件
        output_rou: 输出路由文件
        sample_rate: 采样间隔（秒）
    """
    
    # 1. 读取NGSIM数据
    df = pd.read_csv(ngsim_csv)
    
    # 2. 数据清洗
    df = df.dropna()
    df = df[df['v_Class'] != 0]  # 移除无效数据
    
    # 3. 创建SUMO路网
    # 根据NGSIM道路几何创建边和节点
    create_sumo_net_from_ngsim(df, output_net)
    
    # 4. 创建车辆轨迹
    vehicles = df.groupby('Vehicle_ID')
    
    routes = ET.Element('routes')
    
    for veh_id, veh_data in vehicles:
        # 提取轨迹点
        trajectory = veh_data.sort_values('Global_Time')
        
        # 采样
        trajectory = trajectory.iloc[::sample_rate*10]  # 0.1秒分辨率
        
        # 创建路由
        route = create_route_from_trajectory(trajectory)
        routes.append(route)
    
    # 5. 保存
    tree = ET.ElementTree(routes)
    tree.write(output_rou)
```

```bash
OUTPUT_DIR="data/custom/${SCENARIO_NAME}"

    -n ${OUTPUT_DIR}/${SCENARIO_NAME}.net.xml \

cat > ${OUTPUT_DIR}/${SCENARIO_NAME}.sumocfg << EOF
<?xml version="1.0" encoding="UTF-8"?>
<configuration>
    <input>
        <net-file value="${SCENARIO_NAME}.net.xml"/>
        <route-files value="${SCENARIO_NAME}.rou.xml"/>
    </input>
    <time>
        <begin value="0"/>
        <end value="3600"/>
        <step-length value="0.1"/>
    </time>
    <report>
        <duration-log.statistics value="true"/>
```

### 5.3 Python构建脚本

```python
import subprocess
import yaml
from pathlib import Path

class CustomScenarioBuilder:
    """自定义场景构建器"""
    
    def __init__(self, config_path):
        with open(config_path) as f:
            self.config = yaml.safe_load(f)
        
        self.output_dir = Path(f"data/custom/{self.config['scenario']['name']}")
        self.output_dir.mkdir(parents=True, exist_ok=True)
    
    def build_network(self):
        """构建路网"""
        scenario_type = self.config['scenario']['type']
        
        if scenario_type == 'grid':
            self._build_grid_network()
        elif scenario_type == 'intersection':
            self._build_intersection_network()
        elif scenario_type == 'corridor':
            self._build_corridor_network()
    
    def _build_grid_network(self):
        """构建网格路网"""
        grid_cfg = self.config['grid_config']
        lane_cfg = self.config['lane_config']
        
        cmd = [
            'netconvert', '--grid',
            f"--grid.x-number={grid_cfg['x_number']}",
            f"--grid.y-number={grid_cfg['y_number']}",
            f"--grid.x-length={grid_cfg['x_length']}",
            f"--grid.y-length={grid_cfg['y_length']}",
            f"--default.lanenumber={lane_cfg['num_lanes']}",
            f"--default.speed={lane_cfg['speed']}",
            f"--output-file={self.output_dir}/network.net.xml"
        ]
        
        subprocess.run(cmd, check=True)
    
    def generate_traffic(self):
        """生成交通流"""
        traffic_cfg = self.config['traffic_config']
        
        # 生成随机出行
        cmd = [
            'randomTrips.py',
            '-n', f'{self.output_dir}/network.net.xml',
            '-e', str(traffic_cfg['duration']),
            '-l',
            f"--period={3600/traffic_cfg['flow_rate']}",
            '-o', f'{self.output_dir}/trips.trips.xml'
        ]
        
        subprocess.run(cmd, check=True)
        
        # 生成路由
        cmd = [
            'duarouter',
            '-n', f'{self.output_dir}/network.net.xml',
            '-t', f'{self.output_dir}/trips.trips.xml',
            '--ignore-errors',
            '-o', f'{self.output_dir}/traffic.rou.xml'
        ]
        
        subprocess.run(cmd, check=True)
    
    def create_config_file(self):
        """创建SUMO配置文件"""
        scenario_name = self.config['scenario']['name']
        
        config_content = f"""<?xml version="1.0" encoding="UTF-8"?>
<configuration>
    <input>
        <net-file value="network.net.xml"/>
        <route-files value="traffic.rou.xml"/>
    </input>
    <time>
        <begin value="0"/>
        <end value="{self.config['traffic_config']['duration']}"/>
    </time>
</configuration>
"""
        
        with open(self.output_dir / f"{scenario_name}.sumocfg", 'w') as f:
            f.write(config_content)
    
    def build(self):
        """完整构建流程"""
        print(f"构建场景: {self.config['scenario']['name']}")
        self.build_network()
        self.generate_traffic()
        self.create_config_file()
        print(f"场景构建完成: {self.output_dir}")

# 使用示例
if __name__ == "__main__":
    builder = CustomScenarioBuilder("configs/custom_scenario.yaml")
    builder.build()
```

### 6.1 数据验证

```python
import sumolib
from xml.etree import ElementTree as ET

class DataValidator:
    """SUMO数据验证器"""
    
    def __init__(self, net_file, rou_file):
        self.net_file = net_file
        self.rou_file = rou_file
    
    def validate_network(self):
        """验证路网文件"""
        try:
            net = sumolib.net.readNet(self.net_file)
            
            # 检查节点
            nodes = net.getNodes()
            if len(nodes) == 0:
                return False, "路网无节点"
            
            # 检查边
            edges = net.getEdges()
            if len(edges) == 0:
                return False, "路网无边"
            
            # 检查连通性
            for node in nodes:
                if len(node.getIncoming()) == 0 and len(node.getOutgoing()) == 0:
                    return False, f"孤立节点: {node.getID()}"
            
            # 检查交通信号
            tls = net.getTrafficLights()
            print(f"交通信号数量: {len(tls)}")
            
            return True, "路网验证通过"
            
        except Exception as e:
            return False, f"路网验证失败: {str(e)}"
    
    def validate_routes(self):
        """验证路由文件"""
        try:
            tree = ET.parse(self.rou_file)
            root = tree.getroot()
            
            # 统计车辆
            vehicles = root.findall('.//vehicle')
            flows = root.findall('.//flow')
            
            total_vehicles = len(vehicles)
            for flow in flows:
                veh_per_hour = float(flow.get('vehsPerHour', 0))
                duration = 3600  # 默认1小时
                total_vehicles += veh_per_hour * duration / 3600
            
            print(f"预计车辆总数: {int(total_vehicles)}")
            
            if total_vehicles == 0:
                return False, "路由文件无车辆"
            
            return True, "路由验证通过"
            
        except Exception as e:
            return False, f"路由验证失败: {str(e)}"
    
    def validate_all(self):
        """执行所有验证"""
        results = {}
        
        valid, msg = self.validate_network()
        results['network'] = {'valid': valid, 'message': msg}
        
        valid, msg = self.validate_routes()
        results['routes'] = {'valid': valid, 'message': msg}
        
        return results
```

```python
class DataCleaner:
    """SUMO数据清洗器"""
    
    def __init__(self, net_file):
        self.net_file = net_file
        self.net = sumolib.net.readNet(net_file)
    
    def remove_isolated_nodes(self, output_file):
        """移除孤立节点"""
        # 识别孤立节点
        isolated = []
        for node in self.net.getNodes():
            if (len(node.getIncoming()) == 0 and 
                len(node.getOutgoing()) == 0):
                isolated.append(node.getID())
        
        if isolated:
            print(f"移除孤立节点: {isolated}")
            # 使用netconvert移除
            cmd = [
                'netconvert',
                '-s', self.net_file,
                '--remove-edges', ','.join(isolated),
                '-o', output_file
            ]
            subprocess.run(cmd, check=True)
        else:
            print("无孤立节点")
    
    def remove_short_edges(self, min_length=10, output_file=None):
        """移除过短的边"""
        short_edges = []
        for edge in self.net.getEdges():
            if edge.getLength() < min_length:
                short_edges.append(edge.getID())
        
        if short_edges:
            print(f"移除短边: {short_edges}")
    
    def normalize_coordinates(self, output_file):
        """坐标归一化"""
        # 获取边界
        x_coords = []
        y_coords = []
        
        for node in self.net.getNodes():
            x, y = node.getCoord()
            x_coords.append(x)
            y_coords.append(y)
        
        x_min, x_max = min(x_coords), max(x_coords)
        y_min, y_max = min(y_coords), max(y_coords)
        
        print(f"坐标范围: X=[{x_min}, {x_max}], Y=[{y_min}, {y_max}]")
```

```python
class DataAugmenter:
    """数据增强器"""
    
    def __init__(self, net_file, rou_file):
        self.net_file = net_file
        self.rou_file = rou_file
    
    def scale_traffic_flow(self, scale_factor, output_file):
        """缩放交通流量"""
        tree = ET.parse(self.rou_file)
        root = tree.getroot()
        
        # 缩放flow元素的vehsPerHour
        for flow in root.findall('.//flow'):
            current_rate = float(flow.get('vehsPerHour', 0))
            new_rate = current_rate * scale_factor
            flow.set('vehsPerHour', str(int(new_rate)))
        
        tree.write(output_file)
        print(f"流量缩放: {scale_factor}x")
    
    def add_vehicle_types(self, output_file):
        """添加多种车辆类型"""
        tree = ET.parse(self.rou_file)
        root = tree.getroot()
        
        # 添加vType定义
        types = [
            ('car', 'passenger', 13.89, 1.0),
            ('bus', 'bus', 10.0, 2.5),
            ('truck', 'truck', 11.11, 2.0),
        ]
        
        for name, vclass, max_speed, length in types:
            vtype = ET.Element('vType')
            vtype.set('id', name)
            vtype.set('vClass', vclass)
            vtype.set('maxSpeed', str(max_speed))
            vtype.set('length', str(length))
            root.insert(0, vtype)
        
        tree.write(output_file)
    
    def create_multi_period_routes(self, periods, output_file):
        """创建多时段路由"""
        # periods: [(begin1, end1, flow1), (begin2, end2, flow2), ...]
        
        tree = ET.parse(self.rou_file)
        root = tree.getroot()
        
        # 为每个时段创建flow
        for i, (begin, end, flow) in enumerate(periods):
            flow_elem = ET.Element('flow')
            flow_elem.set('id', f'flow_period_{i}')
            flow_elem.set('begin', str(begin))
            flow_elem.set('end', str(end))
            flow_elem.set('vehsPerHour', str(flow))
            root.append(flow_elem)
        
        tree.write(output_file)
```

### 7.2 集成代码实现

```python
import sumo_rl
from torchrl.envs import EnvBase, TransformedEnv, Compose
from torchrl.envs.transforms import ObservationNorm, RewardScaling, StepCounter
from torchrl.envs.libs.pettingzoo import PettingZooWrapper
from typing import Dict, Any
import yaml

class TrafficRLEnvironment:
    """交通RL环境集成类"""
    
    def __init__(self, config_path: str):
        """
        初始化环境
        
        Args:
            config_path: 配置文件路径
        """
        with open(config_path) as f:
            self.config = yaml.safe_load(f)
        
        self.env = None
        self.transformed_env = None
    
    def create_base_env(self):
        """创建基础SUMO-RL环境"""
        env_cfg = self.config['environment']
        sim_cfg = self.config['simulation']
        
        # 选择单Agent或多Agent
        if env_cfg.get('single_agent', False):
            self.env = sumo_rl.environment.env(
                net_file=env_cfg['net_file'],
                route_file=env_cfg['route_file'],
                use_gui=sim_cfg['use_gui'],
                num_seconds=sim_cfg['num_seconds'],
                delta_time=sim_cfg['delta_time'],
                yellow_time=sim_cfg['yellow_time'],
                min_green=sim_cfg['min_green'],
                max_green=sim_cfg['max_green'],
                reward_fn=self._get_reward_fn()
            )
        else:
            self.env = sumo_rl.parallel_env(
                net_file=env_cfg['net_file'],
                route_file=env_cfg['route_file'],
                use_gui=sim_cfg['use_gui'],
                num_seconds=sim_cfg['num_seconds'],
                delta_time=sim_cfg['delta_time'],
                yellow_time=sim_cfg['yellow_time'],
                min_green=sim_cfg['min_green'],
                max_green=sim_cfg['max_green'],
                reward_fn=self._get_reward_fn()
            )
        
        return self.env
    
    def _get_reward_fn(self):
        """获取奖励函数"""
        reward_type = self.config['reward']['function']
        
        if reward_type == 'queue':
            return 'queue'
        elif reward_type == 'waiting_time':
            return 'waiting_time'
        elif reward_type == 'speed':
            return 'speed'
        else:
            return 'diff-waiting-time'  # 默认
    
    def wrap_for_torchrl(self):
        """包装为TorchRL环境"""
        if self.env is None:
            self.create_base_env()
        
        # PettingZoo包装
        if not self.config['environment'].get('single_agent', False):
            torchrl_env = PettingZooWrapper(self.env)
        else:
            # 单Agent直接使用Gymnasium包装
            from torchrl.envs.libs.gym import GymWrapper
            torchrl_env = GymWrapper(self.env)
        
        # 添加变换
        transforms = []
        
        if self.config['observation'].get('normalize', False):
            transforms.append(ObservationNorm(in_keys=["observation"]))
        
        if 'scale' in self.config['reward']:
            transforms.append(
                RewardScaling(loc=0, scale=self.config['reward']['scale'])
            )
        
        transforms.append(StepCounter())
        
        self.transformed_env = TransformedEnv(
            torchrl_env,
            Compose(*transforms)
        )
        
        # 初始化归一化统计
        if self.config['observation'].get('normalize', False):
            self.transformed_env.transform[0].init_stats(
                num_iter=1000,
                reduce_dim=0
            )
        
        return self.transformed_env
    
    def get_env_info(self) -> Dict[str, Any]:
        """获取环境信息"""
        if self.transformed_env is None:
            self.wrap_for_torchrl()
        
        return {
            'observation_spec': self.transformed_env.observation_spec,
            'action_spec': self.transformed_env.action_spec,
            'reward_spec': self.transformed_env.reward_spec,
            'num_agents': len(self.env.ts_ids) if hasattr(self.env, 'ts_ids') else 1
        }
    
    def test_env(self, num_steps: int = 100):
        """测试环境"""
        if self.transformed_env is None:
            self.wrap_for_torchrl()
        
        print("测试环境...")
        td = self.transformed_env.reset()
        print(f"初始观察: {td}")
        
        for i in range(num_steps):
            # 随机动作
            action = self.transformed_env.action_spec.rand()
            td = self.transformed_env.step(action)
            
            if i % 10 == 0:
                print(f"Step {i}, Reward: {td['reward'].item():.4f}")
            
            if td['done'].any():
                print(f"Episode结束于Step {i}")
                td = self.transformed_env.reset()
        
        print("环境测试完成")
        self.transformed_env.close()

# 使用示例
if __name__ == "__main__":
    # 创建环境
    env_manager = TrafficRLEnvironment("configs/env_config.yaml")
    
    # 获取环境信息
    info = env_manager.get_env_info()
    print(f"环境信息: {info}")
    
    # 测试环境
    env_manager.test_env(num_steps=100)
```

### 7.3 多场景数据管理

```python
from pathlib import Path
from typing import List, Dict
import json

class ScenarioManager:
    """场景管理器"""
    
    def __init__(self, data_root: str = "data"):
        self.data_root = Path(data_root)
        self.scenarios = self._scan_scenarios()
    
    def _scan_scenarios(self) -> Dict[str, Dict]:
        """扫描可用场景"""
        scenarios = {}
        
        # 扫描RESCO场景
        resco_path = self.data_root / "RESCO"
        if resco_path.exists():
            for scenario_dir in resco_path.iterdir():
                if scenario_dir.is_dir():
                    net_files = list(scenario_dir.glob("*.net.xml"))
                    if net_files:
                        scenarios[f"RESCO/{scenario_dir.name}"] = {
                            'net_file': str(net_files[0]),
                            'route_files': [str(f) for f in scenario_dir.glob("*.rou.xml")],
                            'type': 'RESCO'
                        }
        
        # 扫描自定义场景
        custom_path = self.data_root / "custom"
        if custom_path.exists():
            for scenario_dir in custom_path.iterdir():
                if scenario_dir.is_dir():
                    net_files = list(scenario_dir.glob("*.net.xml"))
                    if net_files:
                        scenarios[f"custom/{scenario_dir.name}"] = {
                            'net_file': str(net_files[0]),
                            'route_files': [str(f) for f in scenario_dir.glob("*.rou.xml")],
                            'type': 'custom'
                        }
        
        return scenarios
    
    def list_scenarios(self) -> List[str]:
        """列出所有场景"""
        return list(self.scenarios.keys())
    
    def get_scenario(self, name: str) -> Dict:
        """获取场景信息"""
        return self.scenarios.get(name)
    
    def get_scenario_config(self, name: str, route_idx: int = 0) -> Dict:
        """获取场景配置"""
        scenario = self.get_scenario(name)
        if scenario is None:
            raise ValueError(f"场景不存在: {name}")
        
        route_files = scenario['route_files']
        if route_idx >= len(route_files):
            route_idx = 0
        
        return {
            'name': name,
            'net_file': scenario['net_file'],
            'route_file': route_files[route_idx],
            'type': scenario['type']
        }
    
    def save_catalog(self, output_file: str = "data/scenario_catalog.json"):
        """保存场景目录"""
        with open(output_file, 'w') as f:
            json.dump(self.scenarios, f, indent=2)
        print(f"场景目录已保存: {output_file}")

# 使用示例
if __name__ == "__main__":
    manager = ScenarioManager()
    
    print("可用场景:")
    for scenario in manager.list_scenarios():
        print(f"  - {scenario}")
    
    # 保存目录
    manager.save_catalog()
```
